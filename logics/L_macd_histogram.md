## 八、实战伪代码

```python
# L_macd 战法实战算法

def macd_signal(klines):
    """
    输入: K 线数据（含 MACD: DIF, DEA, 红绿柱）
    输出: 信号类型 + 建议操作
    """
    
    latest = klines[-1]
    prev = klines[-2]
    
    dif = latest['DIF']
    dea = latest['DEA']
    hist = latest['MACD_Hist']  # 红绿柱
    prev_hist = prev['MACD_Hist']
    
    # 判断 0 轴位置
    if dif > 0 and dea > 0:
        # 0 轴以上 - 多头地盘
        if hist > 0 and prev_hist > 0:
            if hist > prev_hist:
                return "多头鼎盛 - 持仓"
            elif hist < prev_hist:
                return "⚠️ 红柱缩短 - 准备清仓"
        elif hist == 0 or (hist <= 0 < prev_hist):
            return "🔴 红柱消失 - 多头终结"
        elif hist < 0 and prev_hist >= 0:
            return "🏆 多翻空 - 最高胜率做空点"
        elif hist < 0:
            if abs(hist) > abs(prev_hist):
                return "绿柱变长 - 空头增强"
            elif abs(hist) < abs(prev_hist):
                return "🏆 绿柱缩短 - 清仓离场"
    
    elif dif < 0 and dea < 0:
        # 0 轴以下 - 空头地盘（对称）
        if hist < 0 and prev_hist < 0:
            if abs(hist) > abs(prev_hist):
                return "空头鼎盛 - 观望"
            elif abs(hist) < abs(prev_hist):
                return "⚠️ 绿柱缩短 - 准备抄底"
        elif hist == 0 or (hist >= 0 > prev_hist):
            return "🟢 绿柱消失 - 空头终结"
        elif hist > 0 and prev_hist <= 0:
            return "🏆 空翻多 - 最高胜率抄底点"
        elif hist > 0:
            if hist > prev_hist:
                return "红柱变长 - 多头增强"
            elif hist < prev_hist:
                return "🏆 红柱缩短 - 清仓离场"
    
    return "无信号"
```
