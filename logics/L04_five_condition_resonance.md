## 六、过滤逻辑（伪代码）

```python
def five_condition_resonance(stock):
    # 5 条件共振
    if not macd_strict_golden_cross(stock): return False  # 1
    if not kdj_strict_golden_cross(stock): return False   # 2
    if stock.kdj_j >= 80: return False                    # 3
    if stock.volume_ratio >= 1.5: return False            # 4
    if stock.close <= stock.ma20: return False            # 5

    # 基本面 4 重过滤
    if stock.price >= 100: return False                   # A
    if stock.market_cap >= 500e8: return False            # B
    if stock.pe_ttm >= 30: return False                   # C
    if stock.net_profit_yoy <= 0: return False            # D

    # 消息面 3 重排除
    if stock.has_profit_warning: return False             # E
    if stock.has_insider_selling: return False            # F
    if stock.has_lockup_expiry: return False              # G

    return True
```
